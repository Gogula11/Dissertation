```latex
% Fabric dyeing figure
\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.6\textwidth]{figures/fabric_dyeing_context}
    \caption{Industrial-scale fabric dyeing vats. Batch-to-batch colour transitions
             incur asymmetric cleaning costs, motivating the PMSP-SDSC formulation
             in \S\ref{sec:formalisation}.}
    \label{fig:fabric_dyeing}
\end{figure}

% Yarn dyeing figure (alternative or additional)
\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.6\textwidth]{figures/yarn_dyeing_context}
    \caption{Industrial yarn dyeing machine. Yarn packages are loaded into the
             vessel and dye liquor is circulated through the packages. Colour
             transitions require cleaning cycles analogous to cleaning between
             fabric batches.}
    \label{fig:yarn_dyeing}
\end{figure}
```
